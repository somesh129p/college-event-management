from flask import Flask, render_template, request, redirect, url_for, session, flash from werkzeug.utils import secure_filename import os app = Flask(__name__) app.secret_key = 'your_secret_key' # Dummy data and functions for demonstration admins = {'admin': 'password'} events = [] @app.route('/admin/login', methods=['GET', 'POST']) def admin_login(): if request.method == 'POST': username = request.form.get('username') password = request.form.get('password') if username in admins and admins[username] == password: session['admin'] = username return redirect(url_for('dashboard')) else: error = 'Invalid Credentials' return render_template('admin_login.html', error=error) return render_template('admin_login.html') @app.route('/admin/dashboard', methods=['GET', 'POST']) def dashboard(): if 'admin' not in session: return redirect(url_for('admin_login')) if request.method == 'POST': # Add Event name = request.form.get("name", "") date = request.form.get("date", "") venue = request.form.get("venue", "") image = request.files.get("image") if name and date and venue and image: filename = secure_filename(image.filename) image.save(os.path.join('static/uploads', filename)) event = { 'name': name, 'date': date, 'venue': venue, 'image': filename } events.append(event) flash('Event added successfully!', 'success') else: flash('Please fill all fields and upload an image.', 'danger') return render_template('dashboard.html', events=events) @app.route('/admin/logout') def admin_logout(): session.pop('admin', None) return redirect(url_for('admin_login')) if __name__ == '__main__': app.run(debug=True)
